% Portadilla ilustrada (.tif original de 2006 centrado a tamaño real 125x185 mm)
\newcommand{\portadillaGrafica}[1]{%
  %%SALTO_PORTADILLA_GRAFICA%%
  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture,overlay]
    \node[inner sep=0pt,anchor=center] at (current page.center) {%
      \includegraphics[width=%%PORTADILLA_ANCHO_MM%%mm,height=%%PORTADILLA_ALTO_MM%%mm,keepaspectratio]{#1}%
    };
  \end{tikzpicture}%
  \null\clearpage
}

% Portadilla tipográfica para secciones nuevas (ej. Anexo editorial A1)
\newcommand{\portadillaTipografica}[2]{%
  %%SALTO_PORTADILLA_TIPOGRAFICA%%
  \thispagestyle{empty}%
  \vspace*{48mm}%
  \begin{center}
    {\fontsize{11}{14}\selectfont #1\par}\vspace{8mm}%
    {\bauhausfont\fontsize{18}{22}\selectfont #2\par}
  \end{center}
  \clearpage
}

% Nivel Parte en el Índice
\renewcommand*\l@part[2]{%
  \addpenalty{-\@highpenalty}%
  \addvspace{%%TOC_ESPACIO_PARTE_PT%%pt}%
  \begingroup
    \parindent \z@ \leftskip \z@ \rightskip \@pnumwidth
    \parfillskip -\@pnumwidth
    \leavevmode
    {\bfseries\fontsize{%%TOC_PARTE_PT%%}{%%TOC_INTERLINEADO_PT%%}\selectfont #1}\hfill
    \hb@xt@\@pnumwidth{\hss{\bauhausfont\fontsize{%%TOC_CAP_PT%%}{%%TOC_INTERLINEADO_PT%%}\selectfont #2}}\par
    \nobreak
  \endgroup
}

% Nivel Capítulo / Ensayo en el Índice
\renewcommand*\l@chapter[2]{%
  \addpenalty{-\@highpenalty}%
  \addvspace{%%TOC_ESPACIO_CAP_PT%%pt}%
  \begingroup
    \parindent \z@ \leftskip 0.8em \rightskip \@tocrmarg
    \parfillskip -\@tocrmarg
    \leavevmode
    {\fontsize{%%TOC_CAP_PT%%}{%%TOC_INTERLINEADO_PT%%}\selectfont #1}%
    %%TOC_RELLENO_CAP%%
    \hb@xt@\@pnumwidth{\hss{\fontsize{%%TOC_CAP_PT%%}{%%TOC_INTERLINEADO_PT%%}\selectfont #2}}\par
  \endgroup
}

% Nivel Subsección (##) en el Índice (si incluir_subsecciones: true)
\renewcommand*\l@section[2]{%
  \begingroup
    \parindent \z@ \leftskip 1.6em \rightskip \@tocrmarg
    \parfillskip -\@tocrmarg
    \leavevmode
    {\itshape\fontsize{%%TOC_SEC_PT%%}{%%TOC_INTERLINEADO_PT%%}\selectfont #1}%
    %%TOC_RELLENO_CAP%%
    \hb@xt@\@pnumwidth{\hss{\fontsize{%%TOC_SEC_PT%%}{%%TOC_INTERLINEADO_PT%%}\selectfont #2}}\par
  \endgroup
}
